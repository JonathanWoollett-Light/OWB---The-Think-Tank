on_actions = {
	# Sets up the Broken Mojave (nf_tnk.62) as soon as MOJ no longer exists — whether TNK wins
	# the war in the Mojave or spawns at game start (by gamerule) with MOJ already gone.
	# Fires once, guarded by the tnkd_broken_mojave_setup country flag.
	on_daily_TNK = {
		effect = {
			if = {
				limit = {
					NOT = { country_exists = MOJ }
					NOT = { has_country_flag = tnkd_broken_mojave_setup }
				}
				set_country_flag = tnkd_broken_mojave_setup
				country_event = nf_tnk.62
			}
		}
	}
}
