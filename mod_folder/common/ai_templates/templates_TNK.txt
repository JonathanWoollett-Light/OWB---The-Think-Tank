# The Think Tank AI division templates.
#
# The Think Tank runs on a trickle of manpower (tnk_lobotomite_conscription only
# grants weekly_manpower, there is no conscription pool), so the generic human
# infantry the base game hands every nation is unaffordable for the AI and it ends
# up never recruiting anything. Robots cost 5 manpower a piece, so the AI is pointed
# at a unique robot mainline instead. See common/ai_strategy/TNK.txt for the ratios.
tnk_templates_mainline = {

	available_for = {
		TNK
	}

	role = unique_mainline

	upgrade_prio = {
		factor = 100
	}

	# Early game: a swarm of cheap scrap/support robots. Always buildable from the
	# starting tech (robot_makeshift_automatons_tech / robot_scurry_tech) and dirt
	# cheap on manpower, so the AI can mass these out on its manpower drip.
	tnk_robot_swarm = {
		reinforce_prio = 1
		custom_icon = 99
		upgrade_prio = {
			factor = 3
			modifier = { # Drop off once we can field proper roboscorpion lines
				factor = 0
				has_tech = robot_protectron_equipment_tech_1
				num_of_military_factories > 15
			}
		}

		target_template = {
			support = {
				cnc_robot = 1
			}
			regiments = {
				support_robot = 6
			}
		}
	}

	# Mid/late game: combat robots (this is what roboscorpion equipment upgrades),
	# stiffened with a few support robots. Used once the Think Tank has the combat
	# robot techs and the factories to pay for them.
	tnk_roboscorpion_line = {
		reinforce_prio = 1
		custom_icon = 101
		upgrade_prio = {
			factor = 1
			modifier = {
				factor = 5
				has_tech = robot_protectron_equipment_tech_1
			}
			modifier = {
				factor = 3
				num_of_military_factories > 20
			}
			modifier = { # Can't build combat robots yet, stick to the swarm
				factor = 0
				NOT = { has_tech = robot_protectron_equipment_tech_1 }
			}
		}

		target_template = {
			support = {
				cnc_robot = 1
				cnc_robot_2 = 1
				anti_air_support = 1
			}
			regiments = {
				combat_robot = 8
				support_robot = 4
			}
		}
	}
}
