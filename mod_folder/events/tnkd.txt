		name = nf_tnk.54.a

            base = 100

		# research instead. CHC and TTM have no creature ladder, so they get a flat

					ahead_reduction = 2
					uses = 2
					category = industry
				}
			}
			else_if = {
				limit = { has_tech = insect_equipment_3 }
				add_tech_bonus = {
					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}
				set_technology = { insect_beast_upgrade_tech_3 = 1 }
			}
			else_if = {
				limit = { has_tech = insect_beast_upgrade_tech_2 }
				set_technology = { insect_beast_upgrade_tech_3 = 1 }
				set_technology = { insect_equipment_3 = 1 }
			}
			else_if = {
				limit = { has_tech = insect_equipment_2 }
				set_technology = { insect_equipment_3 = 1 }
				set_technology = { insect_beast_upgrade_tech_2 = 1 }
			}
			else_if = {
				limit = { has_tech = insect_beast_upgrade_tech_1 }
				set_technology = { insect_beast_upgrade_tech_2 = 1 }
				set_technology = { insect_equipment_2 = 1 }
			}
			else = {
				set_technology = { insect_equipment_2 = 1 }
				set_technology = { insect_beast_upgrade_tech_1 = 1 }
			}
		}
		else_if = {
			limit = { tag = MLT }
			if = {
				limit = { has_tech = amphibious_beast_upgrade_tech_4 }
				add_tech_bonus = {
					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}
				add_tech_bonus = {
					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}
			}
			else_if = {
				limit = { has_tech = amphibious_beast_equipment_king }
				add_tech_bonus = {
					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}
				set_technology = { amphibious_beast_upgrade_tech_4 = 1 }
			}
			else_if = {
				limit = { has_tech = amphibious_beast_upgrade_tech_3 }
				set_technology = { amphibious_beast_upgrade_tech_4 = 1 }
				set_technology = { amphibious_beast_equipment_king = 1 }
			}
			else_if = {
				limit = { has_tech = amphibious_beast_equipment_armor }
				set_technology = { amphibious_beast_equipment_king = 1 }
				set_technology = { amphibious_beast_upgrade_tech_3 = 1 }
			}

				limit = { has_tech = amphibious_beast_upgrade_tech_2 }
				set_technology = { amphibious_beast_upgrade_tech_3 = 1 }
				set_technology = { amphibious_beast_equipment_armor = 1 }

			else_if = {
				limit = { has_tech = amphibious_beast_equipment_2 }
				set_technology = { amphibious_beast_equipment_armor = 1 }
				set_technology = { amphibious_beast_upgrade_tech_2 = 1 }
			}
			else_if = {
				limit = { has_tech = amphibious_beast_upgrade_tech_1 }
				set_technology = { amphibious_beast_upgrade_tech_2 = 1 }
				set_technology = { amphibious_beast_equipment_2 = 1 }

				limit = { has_tech = amphibious_beast_equipment_1 }
				set_technology = { amphibious_beast_equipment_2 = 1 }
				set_technology = { amphibious_beast_upgrade_tech_1 = 1 }
			}
			else = {
				set_technology = { amphibious_beast_equipment_1 = 1 }
			}
		}
		else_if = {
			limit = { tag = TRL }
			if = {
				limit = { has_tech = burrowing_beast_upgrade_tech_4 }
				add_tech_bonus = {
					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}

					bonus = 0.5
					ahead_reduction = 2
					uses = 2
					category = industry
				}
			}

					bonus = 0.5

			}

				limit = { has_tech = burrowing_beast_equipment_glowing }

				set_technology = { burrowing_beast_upgrade_tech_3 = 1 }

				limit = { has_tech = burrowing_beast_upgrade_tech_2 }
				set_technology = { burrowing_beast_upgrade_tech_3 = 1 }
				set_technology = { burrowing_beast_equipment_glowing = 1 }
			}

				limit = { has_tech = burrowing_beast_equipment_2 }

				set_technology = { burrowing_beast_upgrade_tech_2 = 1 }
